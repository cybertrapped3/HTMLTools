function Export-PodcastLogger {
    	param (
        		[Parameter(Mandatory=$true)]
        		[object]$Data,
       		[Parameter(Mandatory=$true)]
        		[string]$Podcast,
        		[string]$DestinationPath = ".",
 		[bool]$setDebug=$false
    	)

	Debug-Code -debugging $setDebug -message "ENTERED FUNCTION: Export-PodcastLogger. PODCAST: $Podcast"

    # 1. Show a quick visual summary in the console
    Write-Host "`n--- Download Summary for $Podcast ---" -ForegroundColor Cyan

    $Data | Select-Object Number,
		Title, 
                          	datePublished, 
                          	@{Name="DescChars"; Expression={$_.Description.Length}} | 
            Format-Table -AutoSize

    try {
        # 2. Resolve the path and generate filename
        $resolvedPath = (Resolve-Path $DestinationPath).Path
        $timestamp    = Get-Date -Format "yyyy-MM-dd_HH-mm-ss"
        $fileName     = "podcasts_" + $Podcast + "_$timestamp.json"
        $fullPath     = Join-Path -Path $resolvedPath -ChildPath $fileName

        # 3. Save to JSON
        $Data | ConvertTo-Json -Depth 10 | Set-Content -Path $fullPath -Encoding utf8 -ErrorAction Stop

        Write-Host "[SUCCESS] Metadata logged to: $fullPath`n" -ForegroundColor Green
    }
    catch {
        Write-Host "[ERROR] Could not save JSON: $($_.Exception.Message)" -ForegroundColor Red
    }
}

function sanitizeString{ #sanitizeString -setDebug $setDebug -InputString $InputString -MaxLength $MaxLength
	param(
		[bool]$setDebug=$false,
		[Parameter(Mandatory=$true)]
		[string]$InputString,
		[int]$MaxLength = 100
	)	

	Debug-Code -debugging $setDebug -message "ENTERED FUNCTION: sanitizeString. STRING: $InputString"

	# 1. Handle Length Safely (Avoid Substring errors if string is too short)
	$shortString = $InputString.Substring(0, [Math]::Min($InputString.Length, $MaxLength))

	Debug-Code -debugging $setDebug -message "Handle Length Safely (Avoid Substring errors if string is too short): $shortString "

	# 2. Convert specific "Keepers" (Change instead of Delete)
	# This says: "Find any |, . or : and replace it with a dash"
	$clean = $shortString -replace '[|:]', ' -' -replace '&', 'AND' -replace '\?', '' -replace "\'", '' -replace ',', '' -replace "\/", '-' -replace "\!", '' -replace "\(", '-' -replace "\)", '-' -replace '"', ''
	$clean = $clean -replace '\s+', ' '  # Reduces multiple spaces to a single space

	Debug-Code -debugging $setDebug -message "Convert specific 'Keepers' (Change instead of Delete): $clean"

	# 3. Define the "Nuclear Option" Regex
	# [IO.Path]::GetInvalidFileNameChars() gets system-level illegal chars (\ / : * ? " < > |)
	$invalidChars = [RegEx]::Escape(-join [IO.Path]::GetInvalidFileNameChars())
	$customUnwanted = '[:#\?\(\)\[\]]' # Add your specific unwanted chars here
	$fullRegex = "[$invalidChars$customUnwanted]"

	Debug-Code -debugging $setDebug -message "Define the 'Nuclear Option' Regex: $clean"

	# 4. Final Scrub
	$clean = $clean -creplace '\P{IsBasicLatin}', '' # Nuke non-Latin (œ, €, ™, etc.)
	$clean = $clean -replace $fullRegex, ''         # Nuke all illegal/unwanted symbols
	$clean = $clean.Trim()                          # Remove leading/trailing whitespace

	Debug-Code -debugging $setDebug -message "Final Scrub: $clean"

	return $clean
}
function sanitizeDescription {
    param(
        [bool]$setDebug=$false,
        [Parameter(Mandatory=$true)]
        [string]$Description
    )    

    if ([string]::IsNullOrWhiteSpace($Description)) { return "function sanitizeDescription: No description provided." }

    # 1. Strip HTML tags (Preserving content between them)
    $cleanDescription = $Description -replace '<[^>]+>', ''

    # 2. Decode HTML Entities (Handles &amp;, &quot;, \u0027, etc.)
    # Note: [System.Net.WebUtility] is often more available than [System.Web.HttpUtility]
    $cleanDescription = [System.Net.WebUtility]::HtmlDecode($cleanDescription)

    # 3. Clean up specialized/broken encoding characters
    # We use a single -replace with a map for efficiency
    $mappings = @{
        'Â|â|' = ''
        '€™'    = "'"
        '€“'    = "-"
        '€¯'    = " "
        '\\u0027' = "'"
    }
    foreach ($pattern in $mappings.Keys) {
        $cleanDescription = $cleanDescription -replace $pattern, $mappings[$pattern]
    }

    # 4. SMART WHITESPACE CLEANUP
    # Replace multiple SPACES/TABS with one space (preserves newlines!)
    $cleanDescription = $cleanDescription -replace '[ \t]+', ' '
    
    # Trim leading/trailing whitespace from the whole block
    $cleanDescription = $cleanDescription.Trim()

    return $cleanDescription
}

		#Debug-Code -debugging $true -message "DOWNLOADING PODCAST $podcast EPISODE $filename"

		#Invoke-WebRequest -Uri $item.enclosure.url -OutFile $filename

		# 2. Attempt Download
		[int]$useBits = 0
		#if (Get-Module -ListAvailable BitsTransfer) {
		if ($useBits -eq 1) {
   			try {
				Write-Host "Using BITS for: $safeTitle" -ForegroundColor Cyan
				# -ErrorAction Stop ensures it jumps to 'catch' if BITS fails to start
				Start-BitsTransfer -Source $item.enclosure.url -Destination $filename -ErrorAction Stop
			}
			catch {
				Write-Warning "BITS failed. Falling back to Invoke-WebRequest..."

			}
		}
		else {
			Write-Host "BitsTransfer module not found. Using Invoke-WebRequest for: $safeTitle" -ForegroundColor Yellow

			}
	}

	Write-Host "THE EPISODE TITLE: $theTitle **************************" -ForegroundColor Green

			Export-PodcastLogger -setDebug $setDebug -Podcast $podcastName -Data $episodes

		} #end of if download -ne no

	}

	Export-PodcastLogger -setDebug $setDebug -Podcast "ALL" -Data $podcasts
}

function wrapItUp { #wrapItUp -clean $true -mp3Player 'H:\' -thisPath 'G:\PODCASTS\download\'
    param(
        [Parameter(Mandatory=$true)]
        [bool]$clean=$true,

        [Parameter(Mandatory=$true)]
        [string]$mp3Player="H:\",

        [Parameter(Mandatory=$true)]
        [string]$thisPath="G:\PODCASTS\download\"
    )

    # Use Join-Path for cleaner, safer path combining
    [string]$mp3Archive = Join-Path $thisPath "archive"
	
    if ($clean) {
        $today = Get-Date -Format "yyyyMMdd"
        $todayDirectory = Join-Path $mp3Archive $today

        Write-Host "Creating archive directory: $todayDirectory" -ForegroundColor Cyan
        
        # Force/No errors if it already exists
        New-Item -ItemType Directory -Path $todayDirectory -Force | Out-Null

        # Move metadata files
        if (Test-Path "$thisPath*.xml") { Move-Item "$thisPath*.xml" -Destination $todayDirectory }
        if (Test-Path "$thisPath*.json") { Move-Item "$thisPath*.json" -Destination $todayDirectory }
    }

    # --- ADJUSTMENT: Conditional MP3 Player Deletion ---
    if (Test-Path "$mp3Player*.mp3") {
        $playerFiles = Get-ChildItem -Path "$mp3Player*.mp3"
        $todayDate = (Get-Date).Date

        foreach ($file in $playerFiles) {
            # Check if the file was created today
            if ($file.CreationTime.Date -eq $todayDate) {
                Write-Host "Warning: '$($file.Name)' was created TODAY." -ForegroundColor Yellow
                
                # Prompt loop until valid input is received
                while ($true) {
                    $choice = Read-Host "Type 'y' to Confirm deletion, 'keep' to bypass, or 'cancel' to stop program"
                    $choice = $choice.Trim().ToLower()

                    if ($choice -eq 'y') {
                        Remove-Item -Path $file.FullName -Verbose
                        break
                    } elseif ($choice -eq 'keep') {
                        Write-Host "Skipping deletion for: $($file.Name)" -ForegroundColor Green
                        break
                    } elseif ($choice -eq 'cancel') {
                        Write-Host "Exiting program gently as requested..." -ForegroundColor Orange
                        return # Hard stop: Exits the entire wrapItUp function immediately
                    } else {
                        Write-Host "Invalid input. Please type 'y' or 'keep'." -ForegroundColor Red
                    }
                }
            } else {
                # If it's an older file, delete it automatically
                Remove-Item -Path $file.FullName -Verbose
            }
        }
    }

    # --- Copy & Archive New Podcasts ---
    if (Test-Path "$thisPath*.mp3") {
        # Copy recently downloaded podcasts to MP3 Player
        Copy-Item "$thisPath*.mp3" -Destination $mp3Player -Verbose

        # Archive recently downloaded podcasts
        [string]$mp3ArchiveMP3 = Join-Path $mp3Archive "mp3"
        
        # Ensure the archive/mp3 directory exists
        New-Item -ItemType Directory -Path $mp3ArchiveMP3 -Force | Out-Null
        
        Move-Item "$thisPath*.mp3" -Destination $mp3ArchiveMP3 -Verbose
    } else {
        Write-Host "No new MP3 files found in $thisPath to copy." -ForegroundColor Yellow
    }
}
